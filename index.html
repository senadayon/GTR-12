function changeRank() { const targetInput = document.getElementById('targetUser'); const rankSelect = document.getElementById('rankValue'); if (!targetInput || !rankSelect) { alert('エラー：ランク変更の入力欄が見つかりません。'); console.error('targetUser または rankValue が見つかりません'); return; } const target = targetInput.value.trim(); const rank = rankSelect.value; if (!target) { alert('ランクを変更するユーザー名を入力してください。'); return; } const validRanks = [ 'Owner', 'Admin', 'Moderator', 'Member', 'Regular' ]; if (!validRanks.includes(rank)) { alert('選択したランクが正しくありません。'); return; } if (!confirm(`${target} のランクを ${rank} に変更しますか？`)) { return; } console.log('ランク変更を送信:', { target, rank }); if (typeof socket === 'undefined' || !socket.connected) { alert('サーバーに接続されていません。ページを再読み込みしてください。'); return; } socket.emit('admin_action', { action: 'rank', target: target, value: rank }); }
